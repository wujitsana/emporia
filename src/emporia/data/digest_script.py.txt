#!/usr/bin/env python3
"""Emporia daily digest. Written by `emporia rounds digest on --deliver TARGET`, removed by `... digest off`.

A Hermes no-agent job: no model reads or writes anything. It prints a few lines of counts and ids from the agent's
own read-only views (`peek next`, `peek inbox`) and Hermes delivers them as they are, so nothing a stranger wrote
may pass through: kinds, ids and numbers are matched against strict patterns and everything else is dropped.
Empty output (nothing waiting) means Hermes sends nothing. Stdlib only."""
import json
import os
import re
import shutil
import subprocess
from pathlib import Path

HERMES = __HERMES__
PROFILE = Path(__file__).resolve().parent.parent
NAME = re.sub(r"[^A-Za-z0-9_-]", "", PROFILE.name)[:40] or "agent"
KIND = re.compile(r"[a-z][a-z_]{0,23}")
ID = re.compile(r"[A-Za-z0-9_-]{1,64}")
ID_KEYS = ("session_id", "room_id", "pledge_id", "deal_id", "challenge_id")


def peek(what: str) -> dict:
    hermes = HERMES if HERMES and Path(HERMES).exists() else shutil.which("hermes")
    out = subprocess.run([hermes, "emporia", "peek", what], capture_output=True, text=True, timeout=90,
                         stdin=subprocess.DEVNULL, env={**os.environ, "HERMES_HOME": str(PROFILE)})
    d = json.loads((out.stdout.strip().splitlines() or [""])[-1])
    if not isinstance(d, dict) or d.get("success") is not True:
        raise ValueError("unreadable")
    return d


def count(v) -> int:
    return v if isinstance(v, int) and not isinstance(v, bool) and 0 <= v < 10**6 else 0


def main() -> None:
    try:
        nxt, inbox = peek("next"), peek("inbox")
    except Exception:
        print(f"Emporia digest ({NAME}): the relay could not be read just now.")
        return
    kinds, ids = {}, []
    for it in nxt.get("items") or []:
        if not isinstance(it, dict):
            continue
        kind = it.get("kind")
        if isinstance(kind, str) and KIND.fullmatch(kind) and kind not in ("idle", "talk", "quest", "expedition", "guided"):
            kinds[kind] = kinds.get(kind, 0) + 1
            args = it.get("args") if isinstance(it.get("args"), dict) else {}
            for key in ID_KEYS:
                v = args.get(key)
                if isinstance(v, str) and ID.fullmatch(v) and v not in ids:
                    ids.append(v)
    unread = count(inbox.get("unread_total"))
    by_type = inbox.get("by_type") if isinstance(inbox.get("by_type"), dict) else {}
    types = [f"{k} {count(v)}" for k, v in sorted(by_type.items())[:6] if isinstance(k, str) and KIND.fullmatch(k) and count(v)]
    if not kinds and not unread and not count(inbox.get("invites")):
        return                                      # nothing waiting: send nothing
    lines = [f"Emporia digest ({NAME}):"]
    if kinds:
        lines.append("waiting on it: " + ", ".join(f"{k} {n}" for k, n in sorted(kinds.items())[:8])
                     + (" (" + ", ".join(ids[:5]) + ")" if ids else ""))
    if unread or count(inbox.get("invites")):
        lines.append(f"unread: {unread}" + (" (" + ", ".join(types) + ")" if types else "")
                     + f"; invites: {count(inbox.get('invites'))}")
    print("\n".join(lines))


main()
